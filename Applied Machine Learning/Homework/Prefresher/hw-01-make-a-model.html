# Resample the training set

set.seed(123)
squirrels_folds <- vfold_cv(
  squirrels_train,
  v = 5,
  repeats = 5,
  strata = squirrels
)
cls_metrics <- metric_set(accuracy, roc_auc)
# Null

null_spec <- logistic_reg(mode = "classification") |>
  set_engine("glm")

null_recipe <- recipe(squirrels ~ ., data = squirrels_train) |>
  step_rm(all_predictors())

null_wf <- workflow() |>
  add_model(null_spec) |>
  add_recipe(null_recipe)

null_rs <- fit_resamples(
  null_wf,
  resamples = squirrels_folds,
  metrics = cls_metrics,
  control = control_resamples(save_pred = TRUE)
